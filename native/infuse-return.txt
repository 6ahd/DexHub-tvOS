
bool CTVOSExternalApp::HandleURL(const std::string& url)
{
  NSURLComponents* components = [NSURLComponents componentsWithString:
      [NSString stringWithUTF8String:url.c_str()]];
  if (![components.scheme isEqualToString:@"dexhub-player"] ||
      (![components.host isEqualToString:@"success"] &&
       ![components.host isEqualToString:@"error"]))
    return false;
  NSMutableDictionary* payload = [NSMutableDictionary dictionary];
  payload[@"status"] = components.host;
  for (NSURLQueryItem* item in components.queryItems)
    if (item.value && ([@[@"token", @"position", @"lastPlayedUrl", @"errorCode"]
        containsObject:item.name]))
      payload[item.name] = item.value;
  if (![payload[@"token"] length])
    return true;
  NSString* home = [NSString stringWithUTF8String:CTVOSFileUtils::GetUserHomeDirectory()];
  [[NSFileManager defaultManager] createDirectoryAtPath:home withIntermediateDirectories:YES
      attributes:nil error:nullptr];
  NSData* data = [NSJSONSerialization dataWithJSONObject:payload options:0 error:nullptr];
  if (![data writeToFile:[home stringByAppendingPathComponent:@"dexhub-infuse-return.json"]
      options:NSDataWritingAtomic error:nullptr])
    CLog::Log(LOGERROR, "DexHub Infuse return could not be saved");
  else
    CLog::Log(LOGINFO, "DexHub Infuse return received ({})", components.host.UTF8String);
  return true;
}
