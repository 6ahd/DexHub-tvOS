
// DexHub Scene Lifecycle for Kodi built with modern tvOS SDKs.
@interface DexHubSceneDelegate : UIResponder <UIWindowSceneDelegate>
@property(nullable, nonatomic, strong) UIWindow* window;
@property(nonatomic) BOOL wasBackgrounded;
@end

@implementation DexHubSceneDelegate

- (void)scene:(UIScene*)scene
    willConnectToSession:(UISceneSession*)session
                 options:(UISceneConnectionOptions*)connectionOptions
{
  if (![scene isKindOfClass:[UIWindowScene class]])
    return;
  XBMCApplicationDelegate* app =
      (XBMCApplicationDelegate*)UIApplication.sharedApplication.delegate;
  if (app.window)
  {
    self.window = app.window;
    self.window.windowScene = (UIWindowScene*)scene;
    [self.window makeKeyAndVisible];
    XBMCController* controller = (XBMCController*)self.window.rootViewController;
    [controller resumeAnimation];
    [controller enterForeground];
  }
  else
  {
    self.window = [[UIWindow alloc] initWithWindowScene:(UIWindowScene*)scene];
    self.window.rootViewController = [XBMCController new];
    app.window = self.window;
    [self.window makeKeyAndVisible];
    [(XBMCController*)self.window.rootViewController startAnimation];
  }
  for (UIOpenURLContext* context in connectionOptions.URLContexts)
    [app application:UIApplication.sharedApplication openURL:context.URL options:@{}];
}

- (void)sceneDidEnterBackground:(UIScene*)scene
{
  self.wasBackgrounded = YES;
  XBMCController* controller = (XBMCController*)self.window.rootViewController;
  [controller pauseAnimation];
  [controller enterBackground];
}

- (void)sceneWillEnterForeground:(UIScene*)scene
{
  if (!self.wasBackgrounded)
    return;
  self.wasBackgrounded = NO;
  XBMCController* controller = (XBMCController*)self.window.rootViewController;
  [controller resumeAnimation];
  [controller enterForeground];
}

- (void)scene:(UIScene*)scene openURLContexts:(NSSet<UIOpenURLContext*>*)URLContexts
{
  XBMCApplicationDelegate* app =
      (XBMCApplicationDelegate*)UIApplication.sharedApplication.delegate;
  for (UIOpenURLContext* context in URLContexts)
    [app application:UIApplication.sharedApplication openURL:context.URL options:@{}];
}
@end
